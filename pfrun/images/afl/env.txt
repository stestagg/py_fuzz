# AFL++'s FrameShift stage (afl-fuzz-frameshift.c) writes learned byte-offset
# relations into the mutated buffer without bounds-checking against its actual
# length. pymutate's AST mutators routinely shrink Python source well below
# what a queue entry's learned relations expect, causing a heap-buffer-overflow
# write in afl-fuzz's own process (confirmed via ASan) that surfaces later as
# an unrelated "corrupted size vs. prev_size" glibc abort. No fix upstream yet;
# disable the stage entirely.
AFL_FRAMESHIFT_DISABLE=1
